% Requires: \usepackage{algorithm} \usepackage{algpseudocode} \usepackage{amsmath}
\begin{algorithm}[ht]
\caption{Grid-search MPC control step (\texttt{MPCController.choose\_action})}
\label{alg:mpc}
\begin{algorithmic}[1]
\Require history $H$ (last measurements), setpoints $r_t$ for targets $t \in \mathcal{T}$,
         prediction models $M_t$, control variables $V$ with bounds
         $[lo_j, hi_j]$ and $\mathrm{range}_j = hi_j - lo_j$,
         move weights $\lambda_j$, previous control $u^{-}$
\Ensure control $u^{\star}$ to be applied in the next step
\State $\mathcal{U} \gets$ all combinations of the control variables discretized
       within their bounds with step $\Delta u$
\State $J(u) \gets 0$ for all $u \in \mathcal{U}$
\ForAll{$t \in \mathcal{T}$}
    \State $\hat{y}_t(u) \gets M_t\!\left(H,\, u \text{ held constant over the horizon } N\right)$
           for all $u \in \mathcal{U}$ \Comment{batched forecast}
    \State $y^{\mathrm{ref}}_t \gets$ trajectory approaching $r_t$ from the current
           measurement (halving the gap each step, $y^{\mathrm{ref}}_t[N] = r_t$)
    \State $e_t(u) \gets \dfrac{1}{N}\sum_{k=1}^{N} \omega_k
           \big| \hat{y}_t(u)[k] - y^{\mathrm{ref}}_t[k] \big|$
           \Comment{MAE, MAPE or ITAE}
    \Statex \hskip\algorithmicindent
           $\omega_k = 1$ (MAE/MAPE) \quad or \quad
           $\omega_k = \dfrac{k}{\frac{1}{N}\sum_{i=1}^{N} i}
           = \dfrac{2k}{N+1}$ (ITAE, time-weighted)
    \State $J(u) \gets J(u) + w_t \, e_t(u)$
\EndFor
\ForAll{$u \in \mathcal{U}$}
    \State $J(u) \gets J(u) + P(u)$,\quad
           $P(u) = \displaystyle\sum_{j \in V} \lambda_j
           \frac{\bigl| u_j - u^{-}_j \bigr|}{\mathrm{range}_j}$
           \Comment{move suppression}
\EndFor
\State $u^{\star} \gets \arg\min_{u \in \mathcal{U}} J(u)$
\State $\mathcal{N} \gets \{ u : J(u) \le J(u^{\star}) + \varepsilon \}$
       \Comment{near-optimal candidates}
\If{$|\mathcal{N}| > 1$}
    \State $u^{\star} \gets$ element of $\mathcal{N}$ closest to $u^{-}$
           \Comment{anti-chattering tie-break}
\EndIf
\State $u^{-} \gets u^{\star}$
\State \Return $u^{\star}$ \Comment{applied for one step; repeat after new measurements}
\end{algorithmic}
\end{algorithm}
